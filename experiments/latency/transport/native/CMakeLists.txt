cmake_minimum_required(VERSION 3.16)
project(shizuka_hal_transport LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(SHIZUKA_TRANSPORT_SANITIZERS "Enable host ASan/UBSan" OFF)
option(SHIZUKA_TRANSPORT_HOST_TESTS "Build the synthetic HAL/DSP contract test" ON)
add_library(shizuka_hal_transport STATIC src/hal_transport.c)
target_include_directories(shizuka_hal_transport PUBLIC include)
target_compile_options(shizuka_hal_transport PRIVATE
  -Wall -Wextra -Wpedantic -Werror -ffreestanding -fno-builtin)
if(SHIZUKA_TRANSPORT_SANITIZERS)
  target_compile_options(shizuka_hal_transport PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(shizuka_hal_transport PUBLIC -fsanitize=address,undefined)
endif()
if(SHIZUKA_TRANSPORT_HOST_TESTS)
  add_executable(hal-transport-test tests/transport_test.c)
  target_compile_options(hal-transport-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(hal-transport-test PRIVATE shizuka_hal_transport)
  enable_testing()
  add_test(NAME hal-transport-contract COMMAND hal-transport-test)
endif()
