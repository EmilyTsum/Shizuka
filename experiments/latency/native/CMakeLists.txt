cmake_minimum_required(VERSION 3.16)
project(shizuka_latency_transform LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SHIZUKA_LATENCY_SANITIZERS "Enable host ASan/UBSan" OFF)
if(SHIZUKA_LATENCY_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
add_library(encoded_ttp STATIC src/encoded_ttp.cpp src/external_config.cpp)
target_include_directories(encoded_ttp PUBLIC include)
add_executable(latency-transform src/main.cpp src/bounded_files.cpp)
target_link_libraries(latency-transform PRIVATE encoded_ttp)
add_executable(transform-test tests/transform_test.cpp)
target_link_libraries(transform-test PRIVATE encoded_ttp)
add_executable(external-config-test tests/external_config_test.cpp)
target_link_libraries(external-config-test PRIVATE encoded_ttp)
enable_testing()
add_test(NAME transform-contract COMMAND transform-test)
add_test(NAME external-config-contract COMMAND external-config-test)
